     1→     1→/**
     2→     2→ * The Rebasser — flagship processor.
     3→     3→ *
     4→     4→ * Pipeline:
     5→     5→ *  1. Analyze the track's bassline (pitch track + envelope, see pitch.ts).
     6→     6→ *     The analysis is cached per file so live-preview re-runs are fast.
     7→     7→ *  2. Transcribe the bassline to semitone notes and shift the WHOLE line
     8→     8→ *     by one global octave so its median centers on the user's target
     9→     9→ *     band. A global shift preserves every interval exactly — the sub
    10→    10→ *     plays the same melody, note for note, just deeper. Two modes:
    11→    11→ *     "calm" follows every note (floor-clamped so nothing sinks below
    12→    12→ *     the band), "extreme" locks onto the bassline's DOMINANT note per
    13→    13→ *     section — a steady in-band sub at full pressure, zero bouncing.
    14→    14→ *     Frame data is applied at the CENTER of each analysis window (see
    15→    15→ *     BassAnalysis.centerOffsetSec) plus a note-confirmation delay
    16→    16→ *     (~1.5 hops): a straddling window can lock onto a new note's attack
    17→    17→ *     up to ~100 ms before the note truly starts, which made the sub
    18→    18→ *     switch pitches before the original bass changed. The delay keeps
    19→    19→ *     the sub on the old note until the new one is actually there.
    20→    20→ *  3. Remove the original low end with a steep high-pass (cached per
    21→    21→ *     cutoff/slope). "Auto" clears the entire detected bassline, not just
    22→    22→ *     the sub band, so the track never plays two bass lines at once.
    23→    23→ *  4. Synthesize the new sub: constant per-note frequency with a short
    24→    24→ *     portamento glide between notes, plus a per-sample attack/release
    25→    25→ *     envelope scaled to the sub period. A line-level anti-collision
    26→    26→ *     guard never leaves the sub's core in unison with original notes
    27→    27→ *     that survive the removal filter — two sines at the same pitch
    28→    28→ *     drift in and out of phase, the classic "two basses wobbling"
    29→    29→ *     artefact. The guard shifts the WHOLE line one octave when needed
    30→    30→ *     (a single consistent move — the old per-note guard made
    31→    31→ *     individual notes leap octaves and the sub audibly bounced).
    32→    32→ *     "Survives" is measured slope-aware: an N dB/oct high-pass leaves
    33→    33→ *     a note below its corner only N·log2(corner/f) dB down, so notes
    34→    34→ *     several semitones below the cutoff still count as audible when
    35→    35→ *     the slope is gentle (or the note is close to the corner).
    36→    36→ *  5. Mix with a sidechain duck — while the sub sounds, the original's
    37→    37→ *     low band dips a few dB so the two layers never fight — then limit
    38→    38→ *     with a slow-release limiter whose release is far longer than a sub
    39→    39→ *     cycle, so it never wobbles the low end.
    40→    40→ */

   118→   118→function synthWaveform(phase: number, kind: RebassWaveform): number {
   119→   119→  switch (kind) {
   120→   120→    case "sine":
   121→   121→      return Math.sin(phase);
   122→   122→    case "saturated": {
   123→   123→      const s = Math.sin(phase);
   124→   124→      const drive = 2.1;
   125→   125→      return Math.tanh(drive * s) / Math.tanh(drive);
   126→   126→    }
   127→   127→    case "triangle":
   128→   128→      return (2 / Math.PI) * Math.asin(Math.sin(phase));
   129→   129→  }
   130→   130→}

   136→   136→interface FileCache {
   137→   137→  analysis: BassAnalysis;
   138→   138→  envMax: number;
   139→   139→  gateDb: number;
   140→   140→  /** gated mean bass RMS (linear) of the original */
   141→   141→  bassRms: number;

   148→   148→const fileCache = new WeakMap<Channels, FileCache>();
   149→   149→
   150→   150→async function getFileCache(
   151→   151→  chs: Channels,
   152→   152→  fs: number,
   153→   153→  onProgress?: ProgressFn
   154→   154→): Promise<FileCache> {
   155→   155→  let c = fileCache.get(chs);
   156→   156→  if (c) return c;
   157→   157→
   158→   158→  const analysis = await analyzeBass(chs, fs, async (f, s) => {
   159→   159→    if (onProgress) await onProgress(f * 0.5, s ?? "Analyzing bass");
   160→   160→  });
   161→   161→
   162→   162→  const frames = analysis.frames;
   163→   163→  let validFrames = 0;
   164→   164→  for (let i = 0; i < frames; i++) {
   165→   165→    if (analysis.f0[i] > 0 && analysis.conf[i] > 0.22) validFrames++;
   166→   166→  }
   167→   167→  const validRatio = frames > 0 ? validFrames / frames : 0;
   168→   168→  const lowConfidence = validFrames < 24 || validRatio < 0.02;
   169→   169→
   170→   170→  let envMax = -Infinity;
   171→   171→  for (let i = 0; i < analysis.envDb.length; i++) {
   172→   172→    if (analysis.envDb[i] > envMax) envMax = analysis.envDb[i];
   173→   173→  }
   174→   174→  if (!isFinite(envMax)) envMax = 0;
   175→   175→  const gateDb = envMax - 32;
   176→   176→
   177→   177→  let bassRmsSum = 0;
   178→   178→  let bassRmsCount = 0;
   179→   179→  for (let i = 0; i < frames; i++) {
   180→   180→    if (analysis.envDb[i] > gateDb) {
   181→   181→      bassRmsSum += Math.pow(10, (analysis.envDb[i] - envMax) / 20);
   182→   182→      bassRmsCount++;
   183→   183→    }
   184→   184→  }
   185→   185→  const bassRms = bassRmsCount > 0 ? bassRmsSum / bassRmsCount : 0.4;
   186→   186→
   187→   187→  c = {
   188→   188→    analysis,
   189→   189→    envMax,
   190→   190→    gateDb,
   191→   191→    bassRms,
   192→   192→    lowConfidence,
   193→   193→    hpKey: null,
   194→   194→    hpChannels: null,
   195→   195→  };
   196→   196→  fileCache.set(chs, c);
   197→   197→  return c;
   198→   198→}

   204→   204→/** nearest semitone index (12-TET; anchor is irrelevant, A1 = 55 Hz) */
   205→   205→function snapToNote(f: number): number {
   206→   206→  return Math.round(12 * Math.log2(f / 55));
   207→   207→}
   208→   208→
   209→   209→const noteToHz = (n: number) => 55 * Math.pow(2, n / 12);
   210→   210→
   211→   211→/** Result of transcribing + mapping the bassline into the sub band. */
   212→   212→export interface BasslineMap {
   213→   213→  /** mapped sub frequency per frame (Hz); 0 where nothing is mapped yet */
   214→   214→  targets: Float32Array;
   215→   215→  /** global octave shift applied to the whole line (−1 … 4) */
   216→   216→  octaveShift: number;
   217→   217→  /** distinct source notes mapped */
   218→   218→  noteCount: number;
   219→   219→  /** span of mapped sub frequencies over confirmed notes (Hz); [0,0] if none */
   220→   220→  subRange: [number, number];
   221→   221→}
   222→   222→
   223→   223→/**
   224→   224→ * Deterministic global mapping. The whole bassline is shifted by ONE
   225→   225→ * octave count (kStar) so its median centers on the band — intervals are
   226→   226→ * preserved exactly, so the sub plays the original melody instead of the
   227→   227→ * per-note "closest octave" zigzag (which turned a 2-semitone step into a
   228→   228→ * 10-semitone leap and sounded like a second, unrelated bass line).
   229→   229→ *
   230→   230→ * Line-level anti-collision guard: if the line's median note would land
   231→   231→ * in unison with an original note that survives the removal high-pass,
   232→   232→ * the WHOLE line shifts one more octave down — one consistent move, no
   233→   233→ * per-note leaps. Two sines at the same pitch but different phase drift
   234→   234→ * in and out of phase — the "two bass frequencies modulating each other"
   235→   235→ * wobble. (Octave-related originals collapsing onto the same sub pitch
   236→   236→ * is normal rebass behaviour and does not beat.)
   237→   237→ *
   238→   238→ * Never below the band: every mapped frequency is held at or above
   239→   239→ * rangeLow (calm) or clamped into the band (extreme), so a 29–34 Hz
   240→   240→ * setting never turns into a 20 Hz rumble small speakers can't even
   241→   241→ * reproduce.
   242→   242→ */
   243→   243→export function mapBasslineNotes(
   244→   244→  analysis: BassAnalysis,
   245→   245→  low: number,
   246→   246→  high: number,
   247→   247→  /** resolved removal cutoff in Hz — originals near/above it survive */
   248→   248→  cutoffHz: number,
   249→   249→  /** removal filter steepness in dB/oct — decides how far below the cutoff a note is still audible */
   250→   250→  removeSlopeDb: number,
   251→   251→  /** extreme: dominant-note sub locked inside the band; calm: melodic */
   252→   252→  mode: RebassMode
   253→   253→): BasslineMap {
   254→   254→  const frames = analysis.frames;
   255→   255→  const fc = Math.sqrt(low * high);
   256→   256→
   257→   257→  // global octave shift: line median → band center. One up-shift is
   258→   258→  // allowed for lines that already sit below the band.
   259→   259→  const med = analysis.medianF0 > 0 ? analysis.medianF0 : 55;
   260→   260→  let kStar = Math.round(Math.log2(med / fc));
   261→   261→  if (kStar < -1) kStar = -1;
   262→   262→  if (kStar > 4) kStar = 4;
   263→   263→
   264→   264→  // sentinel for "no confident bass this frame" — must NOT collide with a
   265→   265→  // real note index: notes below A1 (55 Hz) have negative indices too
   266→   266→  const NO_NOTE = -1000;
   267→   267→
   268→   268→  // snap every confident frame to a semitone index. Sticky deadband: once
   269→   269→  // a note is current, a frame only switches to another note when its f0
   270→   270→  // is clearly closer to it (> 0.85 semitone away). This kills the
   271→   271→  // semitone flip-flop on borderline/detuned basslines — and the pitch
   272→   272→  // sag of a decaying note (synth releases and plucked strings drift
   273→   273→  // flat at the tail) no longer flips the sub off its note before the
   274→   274→  // next note actually arrives.
   275→   275→  const noteIdx = new Int32Array(frames).fill(NO_NOTE);
   276→   276→  let curNote = Number.NaN;
   277→   277→  for (let i = 0; i < frames; i++) {
   278→   278→    const f = analysis.f0[i];
   279→   279→    if (f <= 0 || analysis.conf[i] <= 0.22) continue;
   280→   280→    if (Number.isFinite(curNote)) {
   281→   281→      const semi = Math.abs(12 * Math.log2(f / noteToHz(curNote)));
   282→   282→      if (semi < 0.85) {
   283→   283→        noteIdx[i] = curNote;
   284→   284→        continue;
   285→   285→      }
   286→   286→    }
   287→   287→    curNote = snapToNote(f);
   288→   288→    noteIdx[i] = curNote;
   289→   289→  }
   290→   290→
   291→   291→  // group consecutive frames into note runs (single-frame gaps tolerated)
   292→   292→  interface Run {
   293→   293→    note: number;
   294→   294→    start: number;
   295→   295→    end: number;
   296→   296→    len: number;
   297→   297→  }
   298→   298→  const runs: Run[] = [];
   299→   299→  let i = 0;
   300→   300→  while (i < frames) {
   301→   301→    if (noteIdx[i] === NO_NOTE) {
   302→   302→      i++;
   303→   303→      continue;
   304→   304→    }
   305→   305→    const note = noteIdx[i];
   306→   306→    let j = i;
   307→   307→    let len = 0;
   308→   308→    while (j < frames) {
   309→   309→      if (noteIdx[j] === note) {
   310→   310→        len++;
   311→   311→        j++;
   312→   312→      } else if (
   313→   313→        noteIdx[j] === NO_NOTE &&
   314→   314→        j + 1 < frames &&
   315→   315→        noteIdx[j + 1] === note
   316→   316→      ) {
   317→   317→        j++; // tolerate a single dropout frame inside the run
   318→   318→      } else {
   319→   319→        break;
   320→   320→      }
   321→   321→    }
   322→   322→    runs.push({ note, start: i, end: j, len });
   323→   323→    i = j;
   324→   324→  }
   325→   325→
   326→   326→  // which original notes actually occur, and which of them the removal
   327→   327→  // high-pass leaves audible. Slope-aware threshold: an N dB/oct
   328→   328→  // high-pass leaves a note at f roughly N·log2(cutoff/f) dB down, so a
   329→   329→  // note stays audible (≥ −18 dB, before the sidechain duck takes
   330→   330→  // another 6–12 dB) down to cutoff·2^(−18/N). The old fixed factor
   331→   331→  // (0.945 ≈ one semitone below the corner) ignored the slope: at
   332→   332→  // 48 dB/oct notes 2–4 semitones below the corner still passed at
   333→   333→  // −10…−25 dB, and at 12 dB/oct almost the whole bassline survived.
   334→   334→  // A sub note at (near-)unison with such a residual beats slowly
   335→   335→  // against it — the "two bass frequencies modulating" wobble.
   336→   336→  const AUDIBLE_RESIDUAL_DB = 18;
   337→   337→  const usedNotes = new Set<number>();
   338→   338→  for (const r of runs) if (r.len >= 2) usedNotes.add(r.note);
   339→   339→  const surviveThresh =
   340→   340→    cutoffHz *
   341→   341→    Math.pow(2, -AUDIBLE_RESIDUAL_DB / Math.max(12, removeSlopeDb));
   342→   342→  const surviving = new Set<number>();
   343→   343→  for (const n of usedNotes) {
   344→   344→    if (noteToHz(n) >= surviveThresh) surviving.add(n);
   345→   345→  }
   346→   346→
   347→   347→  // Line-level anti-collision guard: if the line's MEDIAN note still
   348→   348→  // lands in unison with a surviving original at this octave shift (a
   349→   349→  // manual cutoff that doesn't clear the line, or a bassline that
   350→   350→  // already sits inside the band), drop the whole line one more octave.
   351→   351→  // One consistent shift for every note — the previous per-note guard
   352→   352→  // moved individual colliding notes up octaves while their neighbours
   353→   353→  // stayed put, and the sub audibly "bounced" up and down through those
   354→   354→  // sections.
   355→   355→  const medNote = snapToNote(med);
   356→   356→  let kLine = kStar;
   357→   357→  while (kLine < 5 && surviving.has(medNote - 12 * kLine)) kLine++;
   358→   358→  kStar = kLine;
   359→   359→
   360→   360→  const targets = new Float32Array(frames);
   361→   361→  let minHz = Infinity;
   362→   362→  let maxHz = 0;
   363→   363→
   364→   364→  if (mode === "extreme") {
   365→   365→    /* Dominant-note sub: the median of the snapped notes in a ±1 s
   366→   366→     * window per frame, then a debounced switch (the new dominant must
   367→   367→     * hold ~0.4 s before the pitch moves). The sub sits on one steady
   368→   368→     * pitch per section, always inside the band, and pumps with the
   369→   369→     * track's own bass envelope — maximum pressure, and by design it
   370→   370→     * cannot bounce up and down octaves. */
   371→   371→    const W = Math.max(5, Math.round(1.0 / analysis.hopSec));
   372→   372→    const HOLD = Math.max(3, Math.round(0.4 / analysis.hopSec));
   373→   373→    const dom = new Int32Array(frames).fill(NO_NOTE);
   374→   374→    const vals: number[] = [];
   375→   375→    for (let i = 0; i < frames; i++) {
   376→   376→      vals.length = 0;
   377→   377→      for (let j = Math.max(0, i - W); j <= Math.min(frames - 1, i + W); j++) {
   378→   378→        if (noteIdx[j] !== NO_NOTE) vals.push(noteIdx[j]);
   379→   379→      }
   380→   380→      if (vals.length >= 3) {
   381→   381→        vals.sort((a, b) => a - b);
   382→   382→        dom[i] = vals[Math.floor(vals.length / 2)];
   383→   383→      }
   384→   384→    }
   385→   385→    // debounce: hold the current pitch until a different dominant note
   386→   386→    // has persisted for HOLD frames (≈0.4 s)
   387→   387→    let cur = NO_NOTE;
   388→   388→    let cand = NO_NOTE;
   389→   389→    let hold = 0;
   390→   390→    for (let i = 0; i < frames; i++) {
   391→   391→      const d = dom[i];
   392→   392→      if (cur === NO_NOTE && d !== NO_NOTE) {
   393→   393→        cur = d;
   394→   394→      } else if (d !== NO_NOTE && d !== cur) {
   395→   395→        if (d === cand) {
   396→   396→          hold++;
   397→   397→          if (hold >= HOLD) {
   398→   398→            cur = d;
   399→   399→            cand = NO_NOTE;
   400→   400→            hold = 0;
   401→   401→          }
   402→   402→        } else {
   403→   403→          cand = d;
   404→   404→          hold = 1;
   405→   405→        }
   406→   406→      } else if (d === cur) {
   407→   407→        cand = NO_NOTE;
   408→   408→        hold = 0;
   409→   409→      }
   410→   410→      if (cur !== NO_NOTE) {
   411→   411→        const hz = Math.min(high, Math.max(low, noteToHz(cur - 12 * kStar)));
   412→   412→        targets[i] = hz;
   413→   413→        if (hz < minHz) minHz = hz;
   414→   414→        if (hz > maxHz) maxHz = hz;
   415→   415→      }
   416→   416→    }
   417→   417→  } else {
   418→   418→    /* Calm melodic mapping: run-by-run targets like before, but every
   419→   419→     * note is held at or above the band floor. Notes that used to map
   420→   420→     * to 16–25 Hz (inaudible on almost any speaker, and the reason a
   421→   421→     * "29–34 Hz" rebass sounded like 20 Hz or less) now sit AT the
   422→   422→     * floor instead of sinking under it — no octave-fold jumps, just a
   423→   423→     * gentle floor. */
   424→   424→    let lastNote: number | null = null;
   425→   425→    for (const r of runs) {
   426→   426→      const runNote = r.len >= 2 ? r.note : lastNote;
   427→   427→      if (runNote !== null) {
   428→   428→        const hz = Math.max(low, noteToHz(runNote - 12 * kStar));
   429→   429→        for (let f = r.start; f < r.end; f++) targets[f] = hz;
   430→   430→        if (r.len >= 2) {
   431→   431→          lastNote = r.note;
   432→   432→          if (hz < minHz) minHz = hz;
   433→   433→          if (hz > maxHz) maxHz = hz;
   434→   434→        }
   435→   435→      }
   436→   436→    }
   437→   437→  }
   438→   438→  const subRange: [number, number] = maxHz > 0 ? [minHz, maxHz] : [0, 0];
   439→   439→
   440→   440→  // fill gaps between runs with the nearest run's frequency so the glide
   441→   441→  // never has to travel from silence
   442→   442→  let prev = 0;
   443→   443→  for (let f = 0; f < frames; f++) {
   444→   444→    if (targets[f] > 0) prev = targets[f];
   445→   445→    else targets[f] = prev;
   446→   446→  }
   447→   447→  // fill any remaining leading gap with the first mapped note
   448→   448→  let next = 0;
   449→   449→  for (let f = frames - 1; f >= 0; f--) {
   450→   450→    if (targets[f] > 0) next = targets[f];
   451→   451→    else if (next > 0) targets[f] = next;
   452→   452→  }
   453→   453→
   454→   454→  return {
   455→   455→    targets,
   456→   456→    octaveShift: kStar,
   457→   457→    noteCount: usedNotes.size,
   458→   458→    subRange,
   459→   459→  };
   460→   460→}
   461→   461→
   462→   462→/* ------------------------------------------------------------------ */
   463→   463→/* Main processor                                                       */
   464→   464→/* ------------------------------------------------------------------ */
   465→   465→
   466→   466→export async function processRebass(
   467→   467→  chs: Channels,
   468→   468→  fs: number,
   469→   469→  opts: RebassOptions,
   470→   470→  onProgress?: ProgressFn
   471→   471→): Promise<{ channels: Channels; info: RebassInfo }> {
   472→   472→  const low = Math.max(16, Math.min(opts.rangeLow, 200));
   473→   473→  const high = Math.max(low + 2, Math.min(opts.rangeHigh, 220));
   474→   474→  const N = chs[0].length;
   475→   475→
   476→   476→  /* ---------------- 1. analysis (cached) ---------------- */
   477→   477→  const cache = await getFileCache(chs, fs, onProgress);
   478→   478→  const analysis = cache.analysis;
   479→   479→  const frames = analysis.frames;
   480→   480→
   481→   481→  /* ---------------- 2. resolve the removal cutoff ---------------- */
   482→   482→  // "auto" clears the entire detected bassline: p90 of the pitch track
   483→   483→  // plus four semitones of transition-band margin (a 48 dB/oct
   484→   484→  // high-pass leaves a note AT the corner only −12 dB, and the previous
   485→   485→  // two-semitone margin left the top of the bassline just −8…−12 dB
   486→   486→  // down — still loud enough to beat against a sub at the same pitch;
   487→   487→  // +4 semitones pushes the p90 note to ~−27 dB), always at least the
   488→   488→  // top of the target band.
   489→   489→  let cutoffHz: number;
   490→   490→  if (opts.removeBelowHz === "auto") {
   491→   491→    const p90 = analysis.rangeF0[1];
   492→   492→    cutoffHz = Math.min(
   493→   493→      160,
   494→   494→      Math.max(high, p90 > 0 ? p90 * 1.26 : 0, 30)
   495→   495→    );
   496→   496→  } else {
   497→   497→    cutoffHz = Math.max(16, Math.min(opts.removeBelowHz, 250));
   498→   498→  }
   499→   499→
   500→   500→  /* ---------------- 3. transcribe to notes ---------------- */
   501→   501→  if (onProgress) await onProgress(0.52, "Mapping bassline");
   502→   502→  const { targets, octaveShift, noteCount, subRange } = mapBasslineNotes(
   503→   503→    analysis,
   504→   504→    low,
   505→   505→    high,
   506→   506→    cutoffHz,
   507→   507→    opts.removeSlope,
   508→   508→    opts.mode
   509→   509→  );
   510→   510→
   511→   511→  // per-frame amplitude target (0..1), normalized so p95 of active = 1.
   512→   512→  // The gate uses hysteresis (5 dB) and bridges single-frame pitch
   513→   513→  // dropouts so the sub never stutters when the level or the pitch
   514→   514→  // confidence hovers around the gate boundary.
   515→   515→  const envLin = new Float32Array(frames);
   516→   516→  const envDb = analysis.envDb;
   517→   517→  const nEnv = Math.min(frames, envDb.length);
   518→   518→  let gateOn = false;
   519→   519→  for (let f = 0; f < frames; f++) {
   520→   520→    const envOk =
   521→   521→      f < nEnv && envDb[f] > (gateOn ? cache.gateDb - 5 : cache.gateDb);
   522→   522→    let pitchOk = analysis.f0[f] > 0;
   523→   523→    if (!pitchOk && f > 0 && f + 1 < frames) {
   524→   524→      pitchOk = analysis.f0[f - 1] > 0 && analysis.f0[f + 1] > 0;
   525→   525→    }
   526→   526→    gateOn = envOk && pitchOk;
   527→   527→    envLin[f] =
   528→   528→      gateOn && f < nEnv
   529→   529→        ? Math.min(1, dbToGain(Math.min(0, envDb[f] - cache.envMax)))
   530→   530→        : 0;
   531→   531→  }
   532→   532→  const activeVals: number[] = [];
   533→   533→  for (let f = 0; f < frames; f++) {
   534→   534→    if (envLin[f] > 0.02) activeVals.push(envLin[f]);
   535→   535→  }
   536→   536→  activeVals.sort((a, b) => a - b);
   537→   537→  const p95 =
   538→   538→    activeVals.length > 0
   539→   539→      ? activeVals[Math.floor(activeVals.length * 0.95)]
   540→   540→      : 1;
   541→   541→  const norm = p95 > 0.001 ? 1 / p95 : 1;
   542→   542→  for (let f = 0; f < frames; f++) {
   543→   543→    envLin[f] = Math.min(1.15, envLin[f] * norm);
   544→   544→  }
   545→   545→
   546→   546→  /* ---------------- 4. synthesize the sub ---------------- */
   547→   547→  if (onProgress) await onProgress(0.58, "Forging sub-bass");
   548→   548→  const sub = new Float32Array(N);
   549→   549→  // sub presence 0..1 per sample (pre-gain) — drives the sidechain duck
   550→   550→  const envTrack = new Float32Array(N);
   551→   551→  const hopSamples = analysis.hopSec * fs;
   552→   552→  // frame f's estimates describe the CENTER of its analysis window
   553→   553→  // [f·hop, f·hop + 2·centerOffset]. Driving the synth at the window
   554→   554→  // start instead made every note change and envelope movement land
   555→   555→  // roughly half a window (~128 ms) before the actual note.
   556→   556→  const centerSamples = analysis.centerOffsetSec * fs;
   557→   557→  // Note-confirmation delay. Even window-centered, a frame can report a
   558→   558→  // NEW note before its true onset: the 256 ms window straddles the
   559→   559→  // change, and when the old note's tail is quiet the attack transient
   560→   560→  // alone can dominate the autocorrelation — the detected switch then
   561→   561→  // lands up to ~100 ms early and the sub changed pitch before the
   562→   562→  // original bass did. Consuming every frame ~1.5 hops later holds the
   563→   563→  // sub on the old note until the new one is genuinely there (the
   564→   564→  // portamento glide adds its own few ms on top, so the switch lands
   565→   565→  // on the beat, never ahead of it).
   566→   566→  const confirmSamples = Math.min(0.12, analysis.hopSec * 1.5) * fs;
   567→   567→
   568→   568→  const times = FOLLOW_TIMES[opts.follow];
   569→   569→  // attack/release never shorter than a fraction of the sub period —
   570→   570→  // faster envelopes amplitude-modulate deep notes and sound shaky
   571→   571→  const periodSec = 1 / Math.max(1, low);
   572→   572→  const atkSec = Math.max(times.atk, periodSec * 0.5);
   573→   573→  const relSec = Math.max(times.rel, periodSec * 2);
   574→   574→  const atkCoef = Math.exp(-1 / (fs * atkSec));
   575→   575→  const relCoef = Math.exp(-1 / (fs * relSec));
   576→   576→  const glideCoef = Math.exp(-1 / (fs * times.glide));
   577→   577→
   578→   578→  let logF = Math.log(low);
   579→   579→  let envState = 0;
   580→   580→  let phase = 0;
   581→   581→  let subActive = 0;
   582→   582→  let lastYield = 0;
   583→   583→  // cached while the frequency is steady — avoids two Math calls per sample
   584→   584→  let fCur = low;
   585→   585→  let inc = (2 * Math.PI * fCur) / fs;
   586→   586→
   587→   587→  for (let n = 0; n < N; n++) {
   588→   588→    const fpos = Math.max(0, (n - centerSamples - confirmSamples) / hopSamples);
   589→   589→    const i0 = Math.min(frames - 1, Math.floor(fpos));
   590→   590→    const i1 = Math.min(frames - 1, i0 + 1);
   591→   591→    const fr = fpos - i0;
   592→   592→
   593→   593→    // glide toward the current note (log-domain portamento). exp() is only
   594→   594→    // needed while the glide is actually converging
   595→   595→    const ta = targets[i0];
   596→   596→    const tb = targets[i1];
   597→   597→    let hz = 0;
   598→   598→    if (ta > 0 && tb > 0) hz = ta + (tb - ta) * fr;
   599→   599→    else if (ta > 0) hz = ta;
   600→   600→    else if (tb > 0) hz = tb;
   601→   601→    if (hz > 0) {
   602→   602→      const delta = (Math.log(hz) - logF) * (1 - glideCoef);
   603→   603→      if (delta !== 0) {
   604→   604→        logF += delta;
   605→   605→        if (Math.abs(delta) > 1e-10) {
   606→   606→          fCur = Math.min(1000, Math.exp(logF));
   607→   607→          inc = (2 * Math.PI * fCur) / fs;
   608→   608→        }
   609→   609→      }
   610→   610→    }
   611→   611→
   612→   612→    // smooth per-sample envelope
   613→   613→    const ampT = envLin[i0] + (envLin[i1] - envLin[i0]) * fr;
   614→   614→    envState =
   615→   615→      ampT > envState
   616→   616→        ? atkCoef * envState + (1 - atkCoef) * ampT
   617→   617→        : relCoef * envState + (1 - relCoef) * ampT;
   618→   618→
   619→   619→    phase += inc;
   620→   620→    if (phase >= 2 * Math.PI) phase -= 2 * Math.PI;
   621→   621→
   622→   622→    sub[n] = synthWaveform(phase, opts.waveform) * envState;
   623→   623→    envTrack[n] = envState;
   624→   624→    if (envState > 0.02) subActive++;
   625→   625→
   626→   626→    if (onProgress && (n & 0x1fffff) === 0 && n - lastYield > 900000) {
   627→   627→      lastYield = n;
   628→   628→      await onProgress(0.58 + (n / N) * 0.12, "Forging sub-bass");
   629→   629→    }
   630→   630→  }
   631→   631→
   632→   632→  // scale sub level relative to the original bass loudness, with a peak
   633→   633→  // cap that keeps the limiter from riding the sub's own cycles
   634→   634→  let subRms = 0;
   635→   635→  for (let n = 0; n < N; n++) subRms += sub[n] * sub[n];
   636→   636→  subRms = Math.sqrt(subRms / Math.max(1, N));
   637→   637→  const targetRms = cache.bassRms * dbToGain(opts.subLevelDb);
   638→   638→  let subGain = subRms > 1e-6 ? targetRms / subRms : 0;
   639→   639→  const subPeakPre = truePeak([sub]);
   640→   640→  const cap = dbToGain(SUB_PEAK_CAP_DB);
   641→   641→  if (subPeakPre * subGain > cap) subGain = cap / Math.max(subPeakPre, 1e-6);
   642→   642→  for (let n = 0; n < N; n++) sub[n] *= subGain;
   643→   643→  const subPeakDb = gainToDb(truePeak([sub]));
   644→   644→
   645→   645→  /* ---------------- 5. remove original bass + mix ---------------- */
   646→   646→  if (onProgress) await onProgress(0.72, "Removing original bass");
   647→   647→  const cascades = Math.max(1, Math.round(opts.removeSlope / 12));
   648→   648→  const hpKey = `${Math.round(cutoffHz * 10)}|${opts.removeSlope}`;
   649→   649→  let dry: Channels;
   650→   650→  if (cache.hpKey === hpKey && cache.hpChannels) {
   651→   651→    dry = cache.hpChannels;
   652→   652→  } else {
   653→   653→    dry = await filterChannels(
   654→   654→      chs,
   655→   655→      fs,
   656→   656→      { type: "highpass", freq: cutoffHz },
   657→   657→      cascades
   658→   658→    );
   659→   659→    cache.hpKey = hpKey;
   660→   660→    cache.hpChannels = dry;
   661→   661→  }
   662→   662→  // Sidechain duck: while the sub sounds, the original's low band dips so
   663→   663→  // the layers never fight (deeper dip when the sub is set louder:
   664→   664→  // subLevelDb −12…+12 → 6…12 dB).
   665→   665→  const duckDbAmt =
   666→   666→    6 + 6 * Math.min(1, Math.max(0, (opts.subLevelDb + 12) / 12));
   667→   667→  const duckDepth = 1 - dbToGain(-duckDbAmt);
   668→   668→  const duckHz = Math.min(
   669→   669→    180,
   670→   670→    Math.max(90, Math.max(low, cutoffHz) * 3)
   671→   671→  );
   672→   672→  // Linkwitz-Riley split (2 cascaded biquads per side = LR4): the low and
   673→   673→  // high paths sum back to an allpass — flat magnitude, no phase-boost
   674→   674→  // artefacts — so the gain applied to the low path is exact at every
   675→   675→  // frequency. (Subtracting a lowpassed copy instead would boost
   676→   676→  // frequencies where the filter's phase lag turns subtraction into
   677→   677→  // addition.)
   678→   678→  const lowPath = await filterChannels(
   679→   679→    dry,
   680→   680→    fs,
   681→   681→    { type: "lowpass", freq: duckHz },
   682→   682→    2
   683→   683→  );
   684→   684→  const highPath = await filterChannels(
   685→   685→    dry,
   686→   686→    fs,
   687→   687→    { type: "highpass", freq: duckHz },
   688→   688→    2
   689→   689→  );
   690→   690→
   691→   691→  // highPath is a fresh copy we own — accumulate the ducked low band and
   692→   692→  // the sub straight into it
   693→   693→  const out = highPath;
   694→   694→  for (let c = 0; c < out.length; c++) {
   695→   695→    const ch = out[c];
   696→   696→    const lp = lowPath[c];
   697→   697→    for (let n = 0; n < N; n++) {
   698→   698→      const duck = Math.min(1, envTrack[n]);
   699→   699→      ch[n] += lp[n] * (1 - duckDepth * duck) + sub[n];
   700→   700→    }
   701→   701→  }
   702→   702→
   703→   703→  /* ---------------- 5. limiter ---------------- */
   704→   704→  // long release + long lookahead: the gain change per sub cycle stays
   705→   705→  // tiny, so the low end never trembles even when the limiter is working
   706→   706→  if (onProgress) await onProgress(0.85, "Limiting");
   707→   707→  const limiter = await limitChannels(out, fs, {
   708→   708→    ceilingDb: -0.3,
   709→   709→    lookaheadMs: 12,
   710→   710→    releaseMs: 300,
   711→   711→    onProgress: async (f) => {
   712→   712→      if (onProgress) await onProgress(0.85 + f * 0.15, "Limiting");
   713→   713→    },
   714→   714→  });
   715→   715→
   716→   716→  /* ---------------- 6. stats ---------------- */
   717→   717→  const info: RebassInfo = {
   718→   718→    analysis,
   719→   719→    cutoffHz,
   720→   720→    subRangeHz: subRange,
   721→   721→    subActivePct: N > 0 ? (subActive / N) * 100 : 0,
   722→   722→    octaveShift,
   723→   723→    subPeakDb,
   724→   724→    limiter,
   725→   725→    lowConfidence: cache.lowConfidence,
   726→   726→    noteCount,
   727→   727→  };
   728→   728→  if (onProgress) await onProgress(1);
   729→   729→  return { channels: out, info };
   730→   730→}

   734→   734→  return mixdown(chs);
   735→   735→}
